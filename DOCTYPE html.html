    <!-- JAVASCRIPT LOGIKA -->

    <script>

        // Stav aplikace

        let targetCalories = 2000;

        let consumedCalories = 0;

        let foodItems = [];

 

        // Přepínání záložek

        function switchTab(tabId) {

            ['dashboard', 'recipes', 'ai-coach'].forEach(id => {

                document.getElementById(`tab-${id}`).classList.add('hidden');

                document.getElementById(`nav-${id}`).className = "text-gray-500 hover:text-beckoPink pb-1 transition";

            });

            document.getElementById(`tab-${tabId}`).classList.remove('hidden');

            document.getElementById(`nav-${tabId}`).className = tabId === 'ai-coach' || tabId === 'recipes'

                ? "text-beckoPink border-b-2 border-beckoPink pb-1 transition font-semibold"

                : "text-beckoPink border-b-2 border-beckoPink pb-1 transition font-semibold";

        }

 

        // Přidání jídla

        function addFood(event) {

            event.preventDefault();

            const name = document.getElementById('food-name').value;

            const kcal = parseInt(document.getElementById('food-kcal').value);

 

            if (!name || isNaN(kcal)) return;

 

            foodItems.push({ name, kcal });

            consumedCalories += kcal;

 

            updateDashboard();

            document.getElementById('food-form').reset();

        }

 

        // Aktualizace UI přehledu

        function updateDashboard() {

            document.getElementById('target-kcal').innerText = `${targetCalories} kcal`;

            document.getElementById('consumed-kcal').innerText = `${consumedCalories} kcal`;

           

            let remaining = targetCalories - consumedCalories;

            document.getElementById('remaining-kcal').innerText = `${remaining} kcal`;

 

            let progress = Math.min((consumedCalories / targetCalories) * 100, 100);

            document.getElementById('calorie-progress').style.width = `${progress}%`;

 

            const listEl = document.getElementById('food-list');

            if (foodItems.length === 0) {

                listEl.innerHTML = `<li class="text-sm text-gray-400 italic">Zatím žádné zapsané jídlo.</li>`;

            } else {

                listEl.innerHTML = foodItems.map((item, index) => `

                    <li class="flex justify-between items-center bg-gray-50 px-4 py-2 rounded-xl text-sm">

                        <span class="font-medium text-gray-700">${item.name}</span>

                        <div class="flex items-center space-x-3">

                            <span class="text-emerald-600 font-semibold">+${item.kcal} kcal</span>

                            <button onclick="removeFood(${index})" class="text-gray-400 hover:text-red-500 text-xs">✕</button>

                        </div>

                    </li>

                `).join('');

            }

        }

 

        // Odstranění jídla

        function removeFood(index) {

            consumedCalories -= foodItems[index].kcal;

            foodItems.splice(index, 1);

            updateDashboard();

        }

 

        // Lajkování receptů

        function likeRecipe(btn) {

            const countSpan = btn.querySelector('.like-count');

            let count = parseInt(countSpan.innerText);

            countSpan.innerText = count + 1;

            btn.classList.add('scale-125');

            setTimeout(() => btn.classList.remove('scale-125'), 200);

        }

 

        // Modální okno profilu

        function openProfileModal() {

            document.getElementById('profile-modal').classList.remove('hidden');

        }

        function closeProfileModal() {

            document.getElementById('profile-modal').classList.add('hidden');

        }

        function saveProfile() {

            targetCalories = parseInt(document.getElementById('modal-target').value) || 2000;

            let age = document.getElementById('modal-age').value;

            let weight = document.getElementById('modal-weight').value;

           

            document.getElementById('user-details-text').innerHTML = `Cílová hmotnost: <span class="font-semibold text-gray-700">${weight} kg</span> | Věk: <span class="font-semibold text-gray-700">${age} let</span>`;

            closeProfileModal();

            updateDashboard();

        }

 

        // AI Chat simulace

        function handleChatPress(e) {

            if (e.key === 'Enter') sendChatMessage();

        }

 

        function sendChatMessage() {

            const input = document.getElementById('chat-input');

            const text = input.value.trim();

            if (!text) return;

 

            const chatMessages = document.getElementById('chat-messages');

 

            // Zpráva uživatele

            chatMessages.innerHTML += `

                <div class="flex justify-end">

                    <div class="bg-beckoPink text-white p-3 rounded-2xl rounded-tr-none max-w-[80%]">${text}</div>

                </div>

            `;

            input.value = '';

            chatMessages.scrollTop = chatMessages.scrollHeight;

 

            // Simulace odpovědi AI po 1 sekundě

            setTimeout(() => {

                let aiReply = "Rozumím ti. Změna životního stylu je maraton, ne sprint. Hlavně na sebe buď hodný/á a nezapomeň slavit i ty nejmenší pokroky!";

                if (text.toLowerCase().includes('hlad') || text.toLowerCase().includes('nejím')) {

                    aiReply = "Když pociťuješ hlad, zkus se napít sklenice vody nebo sáhnout po bílkovinové svačině (např. tvaroh či ořechy). Tělo si zvyká.";

                } else if (text.toLowerCase().includes('stres') || text.toLowerCase().includes('špatně')) {

                    aiReply = "Mrzí mě, že prožíváš stres. Zkus se teď zhluboka nadechnout a vydechnout. Neúspěch v jídle nebo denní pauza tě nedefinují.";

                }

 

                chatMessages.innerHTML += `

                    <div class="flex justify-start">

                        <div class="bg-gray-100 p-3 rounded-2xl rounded-tl-none max-w-[80%] text-gray-700">${aiReply}</div>

                    </div>

                `;

                chatMessages.scrollTop = chatMessages.scrollHeight;

            }, 1000);

        }

    </script>
